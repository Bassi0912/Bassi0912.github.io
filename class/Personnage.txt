<?php
class Personnage {
    public $id;
    public $nom;
    public $arme;
    public $taille;
    public $type;
    public $statut;
    public $image;

    public function __construct($id, $nom, $arme, $taille, $type, $statut, $image) {
        $this->id = $id;
        $this->nom = $nom;
        $this->arme = $arme;
        $this->taille = $taille;
        $this->type = $type;
        $this->statut = $statut;
        $this->image = $image ?: 'default_image.jpg'; // Image par défaut si absente
    }

    // Récupère un personnage au hasard dans la base de données
    public static function getRandomPersonnage($conn) {
        $result = $conn->query("SELECT * FROM personnages ORDER BY RAND() LIMIT 1");
        $row = $result->fetch_assoc();
        return new Personnage($row['id'], $row['nom'], $row['arme'], $row['taille'], $row['type'], $row['statut'], $row['image']);
    }

    // Récupère un personnage par son nom
    public static function getPersonnageByName($conn, $nom) {
        $query = $conn->prepare("SELECT * FROM personnages WHERE nom = ?");
        $query->bind_param("s", $nom);
        $query->execute();
        $result = $query->get_result();
        if ($result->num_rows > 0) {
            $row = $result->fetch_assoc();
            return new Personnage($row['id'], $row['nom'], $row['arme'], $row['taille'], $row['type'], $row['statut'], $row['image']);
        }
        return null;
    }
}

?>

<?php
session_start();

// Connexion à la base de données
$servername = "localhost";
$username = "12303002";
$password = "080437398EG";
$dbname = "12303002jeu_personnage";

// Créez la connexion
$conn = new mysqli($servername, $username, $password, $dbname);

// Vérifiez la connexion
if ($conn->connect_error) {
    die("Connection failed: " . $conn->connect_error);
}

// Variables pour gérer la tentative et les résultats
$attempt = '';
$correctAttempt = false;
$attemptedCharacter = null;
$randomCharacter = null;

// Si le personnage à deviner n'est pas défini dans la session, on en choisit un aléatoirement
if (!isset($_SESSION['randomCharacter'])) {
    // Récupérer un personnage aléatoire
    $sql = "SELECT * FROM personnages ORDER BY RAND() LIMIT 1";
    $result = $conn->query($sql);

    if ($result->num_rows > 0) {
        $randomCharacter = $result->fetch_assoc();
        $_SESSION['randomCharacter'] = $randomCharacter; // Stocke le personnage dans la session
    } else {
        echo "Aucun personnage trouvé.";
        exit;
    }
} else {
    $randomCharacter = $_SESSION['randomCharacter']; // Récupère le personnage à deviner
}

// Initialiser la liste des tentatives si elle n'existe pas encore
if (!isset($_SESSION['attempts'])) {
    $_SESSION['attempts'] = [];
}

// Initialiser le score si non défini
if (!isset($_SESSION['score'])) {
    $_SESSION['score'] = 100; // Score initial
}

// Traitement de la tentative de l'utilisateur
if ($_SERVER['REQUEST_METHOD'] == 'POST' && !isset($_POST['reset'])) {
    $attempt = $_POST['nom']; // Le nom saisi par l'utilisateur

    // Vérifier si l'utilisateur a entré un personnage valide
    $sqlAttempt = "SELECT * FROM personnages WHERE nom = ?";
    $stmt = $conn->prepare($sqlAttempt);
    $stmt->bind_param('s', $attempt);
    $stmt->execute();
    $resultAttempt = $stmt->get_result();

    if ($resultAttempt->num_rows > 0) {
        // Le personnage existe, récupérer ses données
        $attemptedCharacter = $resultAttempt->fetch_assoc();
        $correctAttempt = (strtolower($attempt) == strtolower($randomCharacter['nom'])); // Vérifier si c'est le bon personnage

        // Réduire le score si la tentative est incorrecte
        if (!$correctAttempt) {
            $_SESSION['score'] -= 5; // Réduction de 10 points par tentative incorrecte
        }
    } else {
        $attemptedCharacter = null;
        $correctAttempt = false;
    }
    $stmt->close();

    // Ajouter la tentative à la session
    $_SESSION['attempts'][] = [
        'attempt' => $attempt, 
        'character' => $attemptedCharacter, 
        'correct' => $correctAttempt
    ];
}

// Si l'utilisateur veut recommencer le jeu
if (isset($_POST['reset'])) {
    // Réinitialiser la session
    session_unset();
    session_destroy();
    header("Location: index.php"); // Rediriger vers la page d'accueil
    exit;
}
?>
